# voltage-controlled-cassette-organ-plugin — distributable product (VST3 / AU / Standalone).
# Thin wrapper around dehli_musikk_sampler_engine (dm::ManifestPluginProcessor).

juce_add_plugin(VoltageControlledCassetteOrgan
    COMPANY_NAME            "DehliMusikk"
    BUNDLE_ID               com.dehlimusikk.voltagecontrolledcassetteorgan
    VERSION                 1.0.0
    PLUGIN_MANUFACTURER_CODE Dhmk
    PLUGIN_CODE              Vcco
    FORMATS                 VST3 AU Standalone
    PRODUCT_NAME            "Voltage Controlled Cassette Organ"
    IS_SYNTH                TRUE
    NEEDS_MIDI_INPUT        TRUE
    NEEDS_MIDI_OUTPUT       FALSE
    IS_MIDI_EFFECT          FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    COPY_PLUGIN_AFTER_BUILD TRUE
    VST3_CATEGORIES         "Instrument" "Sampler"
    AU_MAIN_TYPE            "kAudioUnitType_MusicDevice"
)

target_sources(VoltageControlledCassetteOrgan PRIVATE
    source/Plugin.cpp
)

target_compile_definitions(VoltageControlledCassetteOrgan PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
)

target_link_libraries(VoltageControlledCassetteOrgan PRIVATE
    dehli_musikk_sampler_engine
    juce::juce_audio_utils
    juce::juce_recommended_config_flags
    juce::juce_recommended_warning_flags
)

# Embedded asset bundle (generated by ds-plugin-converter into assets/).
# The manifest is embedded SPLIT (manifest/index.json + modes/ + optional partials/)
# and merged at load — NOT the single manifest.json. Mode slugs and partial names
# share one flat resource namespace, so they must not collide.
file(GLOB VCCO_ASSETS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/manifest/index.json"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/manifest/modes/*.json"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/manifest/partials/*.json"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/samples/*.flac"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/ir/*.flac"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/images/*.png"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/images/*.jpg"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/images/*.jpeg")

if(VCCO_ASSETS)
    juce_add_binary_data(VCCOAssets SOURCES ${VCCO_ASSETS})
    target_link_libraries(VoltageControlledCassetteOrgan PRIVATE VCCOAssets)
    target_compile_definitions(VoltageControlledCassetteOrgan PRIVATE VCCO_HAS_ASSETS=1)
    list(LENGTH VCCO_ASSETS VCCO_ASSET_COUNT)
    message(STATUS "VoltageControlledCassetteOrgan: embedding ${VCCO_ASSET_COUNT} asset file(s) from assets/")
else()
    message(WARNING "VoltageControlledCassetteOrgan: no assets in assets/ — plugin will be silent. "
                    "Generate them with the dmse_convert target first.")
endif()

# Samples ship as a memory-mapped pack (default), NOT compiled into the binary — a 1.4 GB
# library would otherwise make a 1.5 GB executable (slow launch, high RAM). Install the pack
# into the shared support folder the engine reads (keyed by product name). One copy serves
# all formats.
dmse_setup_plugin(VoltageControlledCassetteOrgan "Voltage Controlled Cassette Organ")
